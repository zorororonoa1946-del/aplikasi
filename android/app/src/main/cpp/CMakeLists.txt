cmake_minimum_required(VERSION 3.22.1)
project(nusacrypto LANGUAGES C CXX)
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(BUILD_TESTING OFF CACHE BOOL "Do not build upstream test executables in Android APK" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Link crypto statically into the private JNI library" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
add_subdirectory("${REPOSITORY_ROOT}/.local/boringssl" "${CMAKE_CURRENT_BINARY_DIR}/boringssl" EXCLUDE_FROM_ALL)
add_library(nusacrypto SHARED pbkdf.cpp)
target_compile_features(nusacrypto PRIVATE cxx_std_17)
target_link_libraries(nusacrypto PRIVATE crypto)
target_link_options(nusacrypto PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
